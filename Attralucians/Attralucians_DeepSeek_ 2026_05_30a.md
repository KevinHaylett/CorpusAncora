class AlphonicBase:
    """A finite symbolic base (Alphon).
    An Alphon consists of:
      - An ordered alphabet of digit symbols (strings), from zero to base-1.
      - A successor table:    digit → next digit (and whether a carry occurs)
      - A predecessor table:  digit → previous digit (and whether a borrow 
occurs)
      - An addition table:    (digit, digit) → (result_digit, carry_digit)
      - A subtraction table:  (digit, digit) → (result_digit, borrow_flag)
        where borrow_flag = True means we borrowed from the next position
    All tables are hand-coded as plain Python dicts from symbol strings to
    symbol strings.  NO integer arithmetic is used to derive these values.
    Bases provided: 2 (binary), 10 (decimal), 12 (dozenal).
    """
    def __init__(
        self,
        name: str,
        digits: List[str],
        add_table: dict,       # (d1, d2, carry_in) -> (sum_digit, carry_out)
        zero: str,
        one: str,
    ):
        self.name = name
        self.digits = digits           # ordered list: digits[0] = zero symbol
        self.zero = zero               # the zero symbol
        self.one = one                 # the unit symbol


---

        # Validate add_table keys are symbols in this alphabet.
        for key in add_table:
            d1, d2, c = key
            assert d1 in digits, f"Unknown digit {d1!r} in add_table for {name}"
            assert d2 in digits, f"Unknown digit {d2!r} in add_table for {name}"
            assert c in (zero, one), f"Carry {c!r} must be zero or one symbol"
        self._add_table = add_table    # keyed by (d1, d2, carry_in_symbol)
        # Build successor/predecessor from the add_table as a convenience.
        self._successor = {}   # digit -> (next_digit, carry_symbol)
        self._predecessor = {} # digit -> (prev_digit, borrow_symbol)
        for d in digits:
            result, carry = self._add_table[(d, one, zero)]
            self._successor[d] = (result, carry)
        # Predecessor: find d such that successor(d) = (target, zero)
        inv = {}
        for d, (s, c) in self._successor.items():
            if c == zero:
                inv[s] = d
        # For zero predecessor, borrow is needed.
        for d in digits:
            if d in inv:
                self._predecessor[d] = (inv[d], zero)  # no borrow
            else:
                # d == zero: predecessor wraps (borrows)
                # The "top" digit is the last one
                self._predecessor[d] = (digits[-1], one)

    # Symbol queries
    # ------------------------------------------------------------------
    def is_zero(self, symbol: str) -> bool:
        return symbol == self.zero
    def successor(self, digit: str) -> Tuple[str, str]:
        """Return (next_digit, carry_symbol).  carry_symbol is self.one if
        we wrapped around (i.e. next_digit is zero), else self.zero."""
        return self._successor[digit]


---

    def predecessor(self, digit: str) -> Tuple[str, str]:
        """Return (prev_digit, borrow_symbol).  borrow_symbol is self.one if
        we wrapped (i.e. digit was zero), else self.zero."""
        return self._predecessor[digit]
    def add_digits(self, d1: str, d2: str, carry_in: str) -> Tuple[str, str]:
        """Single-digit addition with carry.
        Returns (sum_digit, carry_out_symbol)."""
        return self._add_table[(d1, d2, carry_in)]

        return self.digits.index(d)

def _sub_digit(minuend: str, subtrahend: str, base: AlphonicBase) -> Tuple[str, 
str]:
    """Single-position symbolic subtraction: minuend - subtrahend.
    Returns (result_digit, borrow_symbol).
    borrow_symbol is base.one if minuend < subtrahend (i.e. we must borrow).
    Method: walk the predecessor chain from minuend for as many steps
    as the positional order of subtrahend, using only table lookups.
    The positional order is used as a *step counter* (control flow), not
    as a numeric value in an arithmetic expression.
    """
    steps = base.digit_order(subtrahend)  # how many predecessor steps 
needed
    current = minuend
    borrow = base.zero
    for _ in range(steps):
        prev, b = base.predecessor(current)
        if not base.is_zero(b):
            # We wrapped — need to borrow from the next position
            borrow = base.one
        current = prev
    return current, borrow
